    <h1>🪙 AR Bounty Clicker</h1>

    <div class="status">Собрано: <span id="score">0</span> / 10</div>

    <button class="btn-claim" id="claimBtn" onclick="startClaiming()">🎯 Забрать всё!</button>

    <div id="log">Подключите MetaMask для начала</div>

<script>

    // ВАШ АДРЕС КОНТРАКТА

    const CONTRACT = "0x5e3b652F8Ef29d50F32Ae2D69920651d10D0227b";

   

    // СПИСОК ВСЕХ 10 ID МОНЕТОК (от demo_001 до demo_010)

    const BOUNTY_IDS = [

        "0x64656d6f5f303031000000000000000000000000000000000000000000000000",

        "0x64656d6f5f303032000000000000000000000000000000000000000000000000",

        "0x64656d6f5f303033000000000000000000000000000000000000000000000000",

        "0x64656d6f5f303034000000000000000000000000000000000000000000000000",

        "0x64656d6f5f303035000000000000000000000000000000000000000000000000",

        "0x64656d6f5f303036000000000000000000000000000000000000000000000000",

        "0x64656d6f5f303037000000000000000000000000000000000000000000000000",

        "0x64656d6f5f303038000000000000000000000000000000000000000000000000",

        "0x64656d6f5f303039000000000000000000000000000000000000000000000000",

        "0x64656d6f5f303130000000000000000000000000000000000000000000000000"

    ];

    const ABI = ["function claim(bytes32 bountyId) external"];

    let signer, contract;

    let collected = 0;

    let collectedIds = [];

    // Инициализация

    window.onload = async () => {

        const board = document.getElementById('gameBoard');

        BOUNTY_IDS.forEach((id, index) => {

            const coin = document.createElement('div');

            coin.className = 'coin';

            coin.innerText = index + 1;

            coin.dataset.id = id;

            coin.onclick = () => collectCoin(coin);

            board.appendChild(coin);

        });

        await connectWallet();

    };

    // Подключение кошелька

    async function connectWallet() {

        if (!window.ethereum) {

            document.getElementById('log').innerText = "❌ Установите MetaMask!";

            return;

        }

        try {

            await window.ethereum.request({ method: 'eth_requestAccounts' });

            const provider = new ethers.providers.Web3Provider(window.ethereum);

            signer = provider.getSigner();

            contract = new ethers.Contract(CONTRACT, ABI, signer);

            document.getElementById('log').innerText = `✅ Подключено: ${await signer.getAddress()}`;

        } catch(e) {

            document.getElementById('log').innerText = "Ошибка подключения";

        }

    }

    // Логика сбора монетки (без блокчейна, просто имитация)

    function collectCoin(coinElement) {

        if (coinElement.classList.contains('collected')) return;

       

        coinElement.classList.add('collected');

        collected++;

        collectedIds.push(coinElement.dataset.id);

        document.getElementById('score').innerText = collected;

        document.getElementById('log').innerText = `🪙 Монета ${collected} собрана!`;

        if (collected === 10) {

            document.getElementById('claimBtn').style.display = 'inline-block';

            document.getElementById('log').innerText = "🎉 Все 10 монет собраны! Нажмите 'Забрать всё!'";

        }

    }

    // ОТПРАВКА В БЛОКЧЕЙН (Забираем все 10 монет)

    async function startClaiming() {

        const btn = document.getElementById('claimBtn');

        btn.disabled = true;

        btn.innerText = "⏳ Отправка...";

        document.getElementById('log').innerText = "Отправка транзакций в сеть...";

        for (let i = 0; i < collectedIds.length; i++) {

            const id = collectedIds[i];

            try {

                document.getElementById('log').innerText = `💰 Клеймим монету ${i+1} из 10...`;

                const tx = await contract.claim(id);

                await tx.wait();

                document.getElementById('log').innerText = `✅ Монета ${i+1} получена!`;

            } catch (e) {

                document.getElementById('log').innerHTML = `<span style="color:red">❌ Ошибка на монете ${i+1}: ${e.message}</span>`;

                btn.disabled = false;

                btn.innerText = "Забрать всё!";

                return;

            }

        }

        document.getElementById('log').innerHTML = "<span style='color:#34d399; font-size:20px;'>🎉 ВСЕ 10 USDC УСПЕШНО ПОЛУЧЕНЫ!</span>";

        btn.innerText = "✅ Всё получено!";

        btn.disabled = true;

    }

</script>
